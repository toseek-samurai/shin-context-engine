# Native platform tests link the same product library as shin-ce/shin-observer.
get_filename_component(observer_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(shin_windows_test target source name)
    add_executable(${target} "${observer_root}/tests/${source}")
    target_link_libraries(${target} PRIVATE shin-observer-native advapi32)
    target_compile_definitions(${target} PRIVATE SHIN_TEST_PROFILES="${observer_root}/../profiles"
        NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_compile_options(${target} PRIVATE ${SHIN_NATIVE_WARNINGS})
    add_test(NAME ${name} COMMAND ${target})
    set_tests_properties(${name} PROPERTIES TIMEOUT 90)
endfunction()
shin_windows_test(shin-observer-windows-storage-tests windows_storage_test.cpp shin-observer-windows-storage)
shin_windows_test(shin-windows-scm-transport-tests windows_scm_transport_test.cpp shin-windows-scm-transport)
target_compile_options(shin-windows-scm-transport-tests PRIVATE /UNDEBUG)
add_test(NAME shin-windows-machine-code COMMAND shin-observer-windows-storage-tests --machine-image "${CMAKE_COMMAND}")
set_tests_properties(shin-windows-machine-code PROPERTIES TIMEOUT 30)
shin_windows_test(shin-observer-windows-runtime-tests windows_runtime_test.cpp shin-observer-windows-runtime)
shin_windows_test(shin-local-changes-tests local_changes_test.cpp shin-local-changes)
shin_windows_test(shin-observer-windows-project-tests windows_project_test.cpp shin-observer-windows-project)
shin_windows_test(shin-observer-windows-worker-tests windows_worker_test.cpp shin-observer-windows-worker)
shin_windows_test(shin-local-worker-process-tests local_worker_process_test.cpp shin-local-worker-process)
