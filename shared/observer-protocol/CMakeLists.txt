cmake_minimum_required(VERSION 3.20)
project(ShinObserverProtocol LANGUAGES CXX)

if(NOT WIN32)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
endif()

option(SHIN_OBSERVER_PROTOCOL_SANITIZERS "Enable ASan/UBSan for shin-observer-protocol" OFF)

add_library(shin-observer-protocol STATIC src/observer_protocol.cpp)
target_compile_features(shin-observer-protocol PUBLIC cxx_std_20)
target_include_directories(shin-observer-protocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(shin-observer-protocol SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party")
if(WIN32)
    target_link_libraries(shin-observer-protocol PUBLIC bcrypt)
    target_compile_definitions(shin-observer-protocol PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
else()
    target_link_libraries(shin-observer-protocol PUBLIC OpenSSL::Crypto)
endif()
if(MSVC)
    set(protocol_warnings /W4 /WX /permissive- /utf-8)
else()
    set(protocol_warnings -Wall -Wextra -Wpedantic -Wmissing-declarations)
endif()
target_compile_options(shin-observer-protocol PRIVATE ${protocol_warnings})

if(SHIN_OBSERVER_PROTOCOL_SANITIZERS)
    target_compile_options(shin-observer-protocol PUBLIC -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(shin-observer-protocol INTERFACE -fsanitize=address,undefined)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(shin-observer-protocol-tests tests/observer_protocol_test.cpp)
    target_link_libraries(shin-observer-protocol-tests PRIVATE shin-observer-protocol)
    target_compile_options(shin-observer-protocol-tests PRIVATE ${protocol_warnings})
    add_test(NAME shin-observer-protocol COMMAND shin-observer-protocol-tests)
endif()
