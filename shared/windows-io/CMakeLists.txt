cmake_minimum_required(VERSION 3.24)
project(ShinWindowsIO LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Windows IO qualification requires a native Windows build")
endif()
add_library(shin-windows-io STATIC src/file.cpp src/transport.cpp src/changes.cpp)
target_include_directories(shin-windows-io PUBLIC include)
target_compile_features(shin-windows-io PUBLIC cxx_std_20)
target_link_libraries(shin-windows-io PRIVATE advapi32 bcrypt)
target_compile_definitions(shin-windows-io PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_definitions(shin-windows-io PRIVATE NTDDI_VERSION=NTDDI_WIN10_RS3)
function(shin_require_windows_image_identity target)
    if(NOT MSVC)
        message(FATAL_ERROR "Windows executing-image qualification currently requires MSVC CodeView identity")
    endif()
    # Keep RSDS in release images too. PDBs are not runtime dependencies.
    # /DEBUG changes optimization defaults, so restore them outside Debug.
    target_link_options(${target} PUBLIC /DEBUG:FULL /INCREMENTAL:NO
        "$<$<NOT:$<CONFIG:Debug>>:/OPT:REF>" "$<$<NOT:$<CONFIG:Debug>>:/OPT:ICF>")
endfunction()
if(MSVC)
    target_compile_options(shin-windows-io PRIVATE /W4 /WX /permissive- /utf-8)
    if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        add_link_options(/MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/application.manifest")
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(shin-windows-file-tests tests/file_test.cpp)
    target_link_libraries(shin-windows-file-tests PRIVATE shin-windows-io)
    target_compile_definitions(shin-windows-file-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    if(MSVC)
        target_compile_options(shin-windows-file-tests PRIVATE /W4 /WX /permissive- /utf-8)
    endif()
    add_test(NAME shin-windows-file COMMAND shin-windows-file-tests)
    set_tests_properties(shin-windows-file PROPERTIES TIMEOUT 30)
endif()
